=====================================================================
  2. ¿QUÉ ME PIDEN?  (el verbo de la consigna)
=====================================================================

  "¿se puede asegurar...?", "¿cumple?", "¿se acepta?",
  "¿se detiene / se cambia / conviene?" .......... ENSAYO (H0, H1, CR, decisión)

  "estimar", "entre qué valores", "como mínimo /
  máximo", "con un X% de confianza" ............... INTERVALO (no lleva H0)

  "¿cuántas unidades más...?" ..................... TAMAÑO DE MUESTRA

  "probabilidad de ... si en realidad es ..." ..... β o 1 − β

  "¿sigue / se ajusta a un modelo...?" ............ CHI: bondad de ajuste

  "¿depende de...?", "¿difieren los %?" ........... CHI: contingencia

  "recta", "por cada ... más", "en función de" .... REGRESIÓN


=====================================================================
  3. ¿QUÉ TEMA ES?  (mirá los DATOS, no solo la frase)
=====================================================================

  Dos variables con UNIDADES DISTINTAS
  (horas y errores, temperatura y ventas) ............ REGRESIÓN (5.7)

  La MISMA variable, a los MISMOS individuos
  dos veces (antes / después) ........................ APAREADAS (5.3)

  La MISMA variable en dos GRUPOS distintos
  (proveedor A y B, turno mañana y tarde) ............ DOS PROMEDIOS (5.2)

  Cuántos cumplen / fallan de un total
  ("de 60 piezas, 6 defectuosas") .................... UNA PROPORCIÓN (5.5)

  Tabla filas × columnas con CANTIDADES de casos ..... CONTINGENCIA (5.6)


=====================================================================
  4. REGLAS
=====================================================================

  H1 = LO QUE ME HACE ACTUAR
    "¿se puede asegurar que X?" ............ H1 = X
    control / recepción / "¿se acepta?" .... H1 = la ALARMA  (p > p0) → "derecha"
    cambio / inversión / "¿conviene?" ...... H1 = la MEJORA  (p < p0) → "izquierda"

  QUIÉN ES 1 Y QUIÉN ES 2  (promedios y apareadas)
    1 = el que la frase dice que tiene MÁS → así la cola es "derecha"
    "A resiste más que B"          → 1 = A
    "el curso aumentó"             → 1 = después
    "el curso redujo los errores"  → 1 = antes (antes tenía más)
    delta0 = el "más de N" de la frase (si no hay número → 0)

  REGRESIÓN: QUIÉN ES x
    x = lo que va después de "por cada" o de "en función de"
    y = lo que responde
    "por cada grado más, las ventas..." → x = temperatura, y = ventas
    beta10 = el número con su signo:
       "aumenta más de 1,5"   →  1.5, "derecha"
       "disminuye más de 1,5" → -1.5, "izquierda"

  COLA = el signo de H1
    >  → "derecha"      <  → "izquierda"      ≠  → "bilateral"
    Fisher (paso previo) y Chi cuadrado: siempre cola derecha.

  TIPO (intervalos)
    "estimar entre..." → "bilateral"   "como mínimo" → "minimo"   "como máximo" → "maximo"

  Confianza = 1 − α        (riesgo 5% → alfa = 0.05, confianza = 0.95)

  --- 5.1 DATOS ------------------------------------------------------

  tabla con intervalos .... n, x, s = resumen_tabla(tabla, "nombre")
  valores sueltos ......... n, x, s = resumen_lista(lista, "nombre")
  te dan n, x̄ y S ......... no hace falta función: usá los números directo


  --- 5.2 DOS PROMEDIOS  (grupo 1 = ver sección 4) -------------------

  1º siempre, Fisher previo:
     metodo = metodo_para_medias(n1, x1, s1, n2, x2, s2, nombres=("1", "2"))

  "¿1 más que 2?":
     ensayo_medias(metodo, delta0, alfa, n1, x1, s1, n2, x2, s2, cola, nombres=("1", "2"))

  "¿1 más de un X% más que 2?"  (la consigna TE DA el %):
     ensayo_medias(metodo, 0, alfa, n1, x1, s1, n2, x2, s2, cola, h=1.10, nombres=("1", "2"))
        "10% más" → h=1.10 · "10% menos" → h=0.90 · con h, delta0 = 0
  σ "históricos" / conocidos:    metodo = "conocidos"   (sin Fisher)

  "diferencia mínima / máxima":
     intervalo_diferencia(metodo, confianza, n1, x1, s1, n2, x2, s2, tipo, nombres=("1", "2"))

  "incremento % mínimo":
     h_limite(metodo, alfa, n1, x1, s1, n2, x2, s2, "derecha", nombres=("1", "2"))


  --- 5.3 APAREADAS  (sin Fisher) ------------------------------------

  ensayo:     ensayo_apareadas(lista1, lista2, delta0, alfa, cola, nombres=("1", "2"))
     ⚠ "reduce / baja el tiempo en más de 3" → antes tenía MÁS:
        ensayo_apareadas(antes, despues, 3, alfa, "derecha")
     ⚠ "aumenta en más de 3" → después tiene MÁS:
        ensayo_apareadas(despues, antes, 3, alfa, "derecha")
  intervalo:  intervalo_apareadas(lista1, lista2, confianza, tipo, nombres=("1", "2"))


  --- 5.4 FISHER COMO PREGUNTA  ("más regular", "reduce el desvío") --

  ensayo_varianzas(phi0, alfa, n1, s1, n2, s2, cola, nombres=("1", "2"))
     phi0 = 1, o 0.8 si dice "en más de un 20%"


  --- 5.5 UNA PROPORCIÓN  (r = cuántos cumplen lo que cuenta p) ------

  "¿cumple? / ¿se acepta?":
     ensayo_proporcion(p0, n, r, alfa, cola)          p0 = el % de la consigna

  "estimar el %":
     intervalo_proporcion(r, n, confianza, tipo)

  "% que no cumple" con tabla:
     r = contar_debajo(tabla, valor)      ("por encima de" → contar_encima)
        ⚠ valor = el corte EN LAS UNIDADES DE LA TABLA ("menos de 14 kg" → 14, NO 0.14)
        ⚠ no hay p0 para cargar: el % es lo que estimás
     intervalo_proporcion(r, n, confianza, tipo)

  "¿cuántas más?" (error e):
     n_para_error_proporcion(r / n, e, confianza)     más = n nuevo − n que tenías

  "probabilidad de ... si en realidad es X%"   (p_estrella = ese X%):
     rc = valor_critico_proporcion(p0, n, alfa, cola)
        ⚠ cola = LA MISMA del ensayo de ese ejercicio
     beta_proporcion(rc, n, p_estrella, cola)
        si es RECHAZAR H0 (detectar, detener, cambiar) → la respuesta es 1 − β
        si es NO rechazar (no detectar, aceptar)       → la respuesta es β


  --- 5.6 CHI CUADRADO  (siempre cola derecha) -----------------------

  "¿sigue un modelo?" (tabla):
     bondad_ajuste(tabla, modelo, alfa)
        modelo = "normal", "gamma", "lognormal", "weibull", "gumbel_min", "gumbel_max"

  con parámetros dados:
     bondad_ajuste(tabla, "normal", alfa, {"mu": valor, "sigma": valor})

  "¿es exponencial?":
     bondad_exponencial(tabla, alfa)        clases abiertas con None: (None, 100, 55)

  probabilidades dadas o iguales:
     bondad_proporciones(categorias, observados, probabilidades, alfa)
        probabilidades = None si son todas iguales

  "¿depende de...?":
     contingencia(tabla, alfa)              una lista por fila, sin totales

  dos proporciones "¿difieren?":
     contingencia([[r1, r2], [n1 - r1, n2 - r2]], alfa)

  dos proporciones con sospecha ("mañana tiene menos que tarde"):
     comparar_proporciones(r1, n1, r2, n2, alfa, "derecha", nombres=("1", "2"))
        grupo 1 = el que debería tener el % MÁS GRANDE

  "prob. de detectar X% más":
     potencia_dos_proporciones(r1, n1, r2, n2, alfa, "derecha", diferencia, nombres=("1", "2"))


  --- 5.7 REGRESIÓN  (x e y = ver sección 4) -------------------------

  recta + validar + % explicado:
     regresion_simple(x, y, alfa, nombres=("X", "Y"))
        el print viene en orden:  HALLAR LA RECTA → VALIDAR → QUÉ % SE EXPLICA

  "¿baja / sube en más de V?":
     ensayo_pendiente(x, y, beta10, alfa, cola, nombres=("X", "Y"))

  "contribución marginal":
     intervalo_pendiente(x, y, confianza, nombres=("X", "Y"))

  "fuerza de la relación" (ρ):
     intervalo_correlacion(x, y, confianza, nombres=("X", "Y"))

  y PROMEDIO para x0 ("los operarios con 16 horas"):
     estimar_media_regresion(x, y, x0, confianza, nombres=("X", "Y"))

  y de UN caso con x0 ("un operario", "un empleado nuevo",
  "si MAÑANA trabaja a...", "la semana que viene", un día concreto):
     predecir_regresion(x, y, x0, confianza, nombres=("X", "Y"))
        → da más ancho que el del promedio
